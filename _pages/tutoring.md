<script>
  (function () {
    var panels = document.querySelectorAll('.lang-panel');
    var buttons = document.querySelectorAll('.lang-btn');
    if (!panels.length || !buttons.length) return;

    function setLanguage(lang) {
      panels.forEach(function (panel) {
        panel.classList.toggle('active', panel.getAttribute('data-lang') === lang);
      });
      buttons.forEach(function (button) {
        var isActive = button.getAttribute('data-lang-target') === lang;
        button.classList.toggle('active', isActive);
        button.classList.toggle('btn-primary', isActive);
        button.classList.toggle('btn-outline-primary', !isActive);
        button.setAttribute('aria-pressed', isActive ? 'true' : 'false');
      });
      localStorage.setItem('tutoring_lang', lang);
    }

    buttons.forEach(function (button) {
      button.addEventListener('click', function () {
        setLanguage(button.getAttribute('data-lang-target'));
      });
    });

    var preferred = localStorage.getItem('tutoring_lang') || 'en';
    if (preferred !== 'fr') {
      preferred = 'en';
    }
    setLanguage(preferred);
  })();
</script>
